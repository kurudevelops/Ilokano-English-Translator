# ================      meaning of tags        ========================= #
# Pronoun = Pronoun (general category)
# PP = Personal Pronoun (I, you, he, she, we, they)
# PP$ = Possessive Pronoun (my, your, his, her, our, their)
# PPS = Subject Pronoun (I, you, he, she, we, they)
# PPO = Object Pronoun (me, you, him, her, us, them)
# PRP = Reflexive Pronoun (myself, yourself, himself, etc.)
# WP = Wh-Pronoun (who, whom, what, which)
# N = Noun (common)
# NP = Noun Phrase
# ProperNoun = Proper Noun (names)
# V = Verb (base form)
# VBD = Verb (past tense)
# VBG = Verb (gerund/present participle)
# VBN = Verb (past participle)
# VP = Verb Phrase
# Adj = Adjective
# Adv = Adverb
# Det = Determiner (the, a, this, those)
# Prep = Preposition (in, on, at, with)
# Conj = Conjunction (and, or, but)
# Aux = Modal Auxiliary (can, could, will, would)
# Neg = Negation (not, don't)
